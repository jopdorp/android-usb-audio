cmake_minimum_required(VERSION 3.22)
project(AndroidUsbAudio LANGUAGES CXX)

if(NOT ANDROID AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "AndroidUsbAudio uses the Android/Linux USB filesystem API")
endif()

find_package(Threads REQUIRED)
add_library(android_usb_audio STATIC UsbAudioDescriptors.cpp UsbAudioDevice.cpp)
target_include_directories(android_usb_audio PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(android_usb_audio PUBLIC cxx_std_20)
target_link_libraries(android_usb_audio PRIVATE Threads::Threads)
set_target_properties(android_usb_audio PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_EXTENSIONS OFF)

set(default_tests OFF)
if(PROJECT_IS_TOP_LEVEL AND NOT ANDROID)
    set(default_tests ON)
endif()
option(ANDROID_USB_AUDIO_BUILD_TESTS "Build the host USB audio tests" ${default_tests})
if(ANDROID_USB_AUDIO_BUILD_TESTS)
    enable_testing()
    add_executable(android_usb_audio_tests tests/UsbAudioTests.cpp)
    target_link_libraries(android_usb_audio_tests PRIVATE android_usb_audio)
    if(NOT ANDROID AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_definitions(android_usb_audio_tests PRIVATE ANDROID_USB_AUDIO_TEST_USBFS)
        target_link_options(android_usb_audio_tests PRIVATE "LINKER:--wrap=ioctl" "LINKER:--wrap=poll")
    endif()
    add_test(NAME android_usb_audio_tests COMMAND android_usb_audio_tests)
endif()
